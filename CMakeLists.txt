cmake_minimum_required(VERSION 3.20)
project(linux_process_monitor VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
function(monitor_warnings target)
  target_compile_options(${target} PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
endfunction()
add_library(monitor_core src/parsing.cpp src/metrics.cpp src/system_monitor.cpp src/process_monitor.cpp src/view_logic.cpp src/presentation.cpp src/sampling.cpp)
monitor_warnings(monitor_core)
target_compile_features(monitor_core PUBLIC cxx_std_20)
target_include_directories(monitor_core PUBLIC include)
find_package(Curses REQUIRED)
add_library(monitor_ui src/terminal_ui.cpp)
target_include_directories(monitor_ui PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(monitor_ui PUBLIC monitor_core PRIVATE ${CURSES_LIBRARIES})
monitor_warnings(monitor_ui)
add_executable(linux-process-monitor src/main.cpp)
target_link_libraries(linux-process-monitor PRIVATE monitor_ui)
monitor_warnings(linux-process-monitor)
include(CTest)
if(BUILD_TESTING)
  add_executable(monitor_tests tests/tests.cpp)
  target_link_libraries(monitor_tests PRIVATE monitor_core)
  monitor_warnings(monitor_tests)
  add_test(NAME core COMMAND monitor_tests)
  add_test(NAME live COMMAND monitor_tests --live)
  set_tests_properties(live PROPERTIES LABELS integration)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_executable(terminal_fixture tests/terminal_fixture.cpp)
    target_link_libraries(terminal_fixture PRIVATE monitor_ui)
    monitor_warnings(terminal_fixture)
    add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_test.py
      $<TARGET_FILE:linux-process-monitor> $<TARGET_FILE:terminal_fixture>)
    set_tests_properties(terminal PROPERTIES LABELS integration TIMEOUT 45)
  endif()
endif()
